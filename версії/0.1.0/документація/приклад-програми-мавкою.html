Початок Завантажити Приклад програми Мавкою
Полотно Кнопки Колір
ЛіМавка0.1.0

Приклад програми Мавкою

кольори = [
  Колір.БІЛИЙ,
  Колір.ЧЕРВОНИЙ,
  Колір.ЗЕЛЕНИЙ,
  Колір.СИНІЙ,
]

г = Полотно.Ш/2
в = Полотно.В/2
р = 20
пк = 0
колір = кольори[пк]

запущено = дійсне

поки запущено
  Полотно.заповнити(Колір.ЧОРНИЙ)
  Полотно.круг(г, в, р, колір)
  Полотно.відмалювати() ;; запланувати відмалювання

  оновити_кнопки() ;; оновити стан кнопок

  якщо Кнопки.вверх.нажато
    в -= 1
  кінець
  якщо Кнопки.вниз.нажато
    в += 1
  кінець
  якщо Кнопки.вліво.нажато
    г -= 1
  кінець
  якщо Кнопки.вправо.нажато
    г += 1
  кінець

  якщо Кнопки.а.нажато
    р += 1
  кінець
  якщо Кнопки.д.нажато
    р -= 1
  кінець

  якщо Кнопки.г.щойно_нажато
    пк += 1
    якщо пк == кольори.довжина
      пк = 0
    кінець

    колір = кольори[пк]
  кінець

  якщо Кнопки.б.щойно_відпущено
    запущено = недійсне
  кінець
кінець
